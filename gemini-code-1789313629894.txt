    <script>
        // Data & Security
        let isLoggedIn = false;

        // Get saved PIN or set default '9876'
        function getStoredPin() {
            return localStorage.getItem('admin_pin') || "9876";
        }

        let driversData = [
            { id: 1, name: "Karim Hossain", category: "Driver", phone: "+880 1711-000000", vehicle: "Covered Van (DH-11-1234)", status: "Active" },
            { id: 2, name: "Anisur Rahman", category: "Manager", phone: "+880 1511-000000", vehicle: "", status: "Active" },
            { id: 3, name: "Rahim Uddin", category: "Helper", phone: "+880 1811-000000", vehicle: "Pickup Truck (DH-55-5678)", status: "Active" },
            { id: 4, name: "Shafiqul Islam", category: "Staff", phone: "+880 1911-000000", vehicle: "Heavy Truck (DH-99-9876)", status: "Active" },
            { id: 5, name: "Jamal Uddin", category: "Loader", phone: "+880 1611-000000", vehicle: "", status: "Active" }
        ];

        // Init App
        document.addEventListener("DOMContentLoaded", () => {
            renderDriversTable(driversData);
            updateDashboardStats();
            document.getElementById('current-year').innerText = new Date().getFullYear();
        });

        // Toggle Expandable Sections inside Admin Panel
        function toggleSection(sectionId, iconId) {
            const section = document.getElementById(sectionId);
            const icon = document.getElementById(iconId);
            
            if (section.classList.contains('hidden')) {
                section.classList.remove('hidden');
                icon.classList.add('rotate-180');
            } else {
                section.classList.add('hidden');
                icon.classList.remove('rotate-180');
            }
        }

        // Tab Switching
        function switchTab(tabName) {
            ['drivers', 'settings'].forEach(t => {
                const btn = document.getElementById(`tab-${t}`);
                if (btn) {
                    btn.classList.remove('text-indigo-700', 'bg-white', 'shadow-sm');
                    btn.classList.add('text-slate-600');
                }
                const content = document.getElementById(`content-${t}`);
                if (content) content.classList.add('hidden');
            });

            const activeBtn = document.getElementById(`tab-${tabName}`);
            activeBtn.classList.add('text-indigo-700', 'bg-white', 'shadow-sm');
            activeBtn.classList.remove('text-slate-600');
            
            document.getElementById(`content-${tabName}`).classList.remove('hidden');

            if (tabName === 'settings') {
                renderAdminDeleteList();
            }
        }

        // Lock & Admin Handlers
        function openSettings() {
            if (isLoggedIn) {
                switchTab('settings');
            } else {
                showLoginModal();
            }
        }

        function showLoginModal() {
            document.getElementById('password-modal').classList.remove('hidden');
            document.getElementById('admin-pass-input').value = '';
            document.getElementById('pass-error').classList.add('hidden');
            setTimeout(() => document.getElementById('admin-pass-input').focus(), 100);
        }

        function closePasswordModal() {
            document.getElementById('password-modal').classList.add('hidden');
        }

        function handlePasswordSubmit(e) {
            e.preventDefault();
            const inputPass = document.getElementById('admin-pass-input').value;
            
            if (inputPass === getStoredPin()) {
                isLoggedIn = true;
                closePasswordModal();
                updateAuthUI();
                switchTab('settings');
            } else {
                document.getElementById('pass-error').classList.remove('hidden');
            }
        }

        // Change Admin PIN Handler
        function handleChangePassword(e) {
            e.preventDefault();
            const curr = document.getElementById('curr-pin-input').value;
            const newPin = document.getElementById('new-pin-input').value;
            const confirmPin = document.getElementById('confirm-pin-input').value;

            if (curr !== getStoredPin()) {
                alert("Incorrect current PIN!");
                return;
            }

            if (newPin !== confirmPin) {
                alert("New PIN and Confirm PIN do not match!");
                return;
            }

            if (newPin.trim().length < 4) {
                alert("New PIN must be at least 4 digits/characters long.");
                return;
            }

            localStorage.setItem('admin_pin', newPin);
            alert("Security PIN updated successfully!");
            
            document.getElementById('curr-pin-input').value = '';
            document.getElementById('new-pin-input').value = '';
            document.getElementById('confirm-pin-input').value = '';
            
            // Auto Collapse after success
            toggleSection('password-section', 'password-icon');
        }

        function logout() {
            isLoggedIn = false;
            updateAuthUI();
            switchTab('drivers');
        }

        function updateAuthUI() {
            const loginBtn = document.getElementById('login-btn');
            const userLoggedIn = document.getElementById('user-logged-in');

            if (isLoggedIn) {
                loginBtn.classList.add('hidden');
                userLoggedIn.classList.remove('hidden');
                userLoggedIn.classList.add('flex');
            } else {
                loginBtn.classList.remove('hidden');
                userLoggedIn.classList.add('hidden');
                userLoggedIn.classList.remove('flex');
            }
        }

        // Helper function for Straight & Uniform Category Badges
        function getCategoryBadgeClass(category) {
            switch(category) {
                case 'Manager':
                    return 'bg-amber-100 text-amber-800 border-amber-200';
                case 'Helper':
                    return 'bg-emerald-100 text-emerald-800 border-emerald-200';
                case 'Staff':
                    return 'bg-purple-100 text-purple-800 border-purple-200';
                case 'Loader':
                    return 'bg-orange-100 text-orange-800 border-orange-200';
                case 'Driver':
                default:
                    return 'bg-blue-100 text-blue-800 border-blue-200';
            }
        }

        // Render Table Data
        function renderDriversTable(data) {
            const tbody = document.getElementById('driver-table-body');
            if (data.length === 0) {
                tbody.innerHTML = `
                    <tr>
                        <td colspan="5" class="p-8 text-center text-slate-400">
                            No matching records found.
                        </td>
                    </tr>`;
                return;
            }

            tbody.innerHTML = data.map(d => `
                <tr class="hover:bg-slate-50 transition-colors">
                    <td class="p-4 pl-6 font-bold text-slate-900 flex items-center gap-3">
                        <div class="w-8 h-8 rounded-full bg-indigo-100 text-indigo-700 flex items-center justify-center font-extrabold text-xs shrink-0">
                            ${d.name.charAt(0)}
                        </div>
                        <span class="truncate">${d.name}</span>
                    </td>
                    <td class="p-4 whitespace-nowrap">
                        <span class="inline-block w-24 text-center py-1 text-xs font-bold rounded-lg border leading-tight ${getCategoryBadgeClass(d.category)}">
                            ${d.category || 'Driver'}
                        </span>
                    </td>
                    <td class="p-4 font-mono font-extrabold text-lg text-slate-900 tracking-wide whitespace-nowrap">${d.phone}</td>
                    <td class="p-4 text-slate-700 font-medium whitespace-nowrap">${d.vehicle ? d.vehicle : '<span class="text-slate-400 font-normal">—</span>'}</td>
                    <td class="p-4 pr-6 whitespace-nowrap">
                        <span class="inline-flex items-center gap-1.5 px-3 py-1 rounded-full text-xs font-bold bg-emerald-100 text-emerald-800">
                            <span class="w-1.5 h-1.5 rounded-full bg-emerald-500"></span>
                            ${d.status}
                        </span>
                    </td>
                </tr>
            `).join('');
        }

        // Render Admin Delete Section with Search Support
        function renderAdminDeleteList() {
            const container = document.getElementById('admin-delete-list');
            const searchQuery = (document.getElementById('delete-search-input')?.value || '').toLowerCase().trim();

            const filteredData = driversData.filter(d => 
                d.name.toLowerCase().includes(searchQuery) ||
                d.phone.includes(searchQuery) ||
                (d.category && d.category.toLowerCase().includes(searchQuery)) ||
                (d.vehicle && d.vehicle.toLowerCase().includes(searchQuery))
            );

            if (filteredData.length === 0) {
                container.innerHTML = `<p class="text-center text-slate-400 py-4 text-xs">No matching records found to delete.</p>`;
                return;
            }

            container.innerHTML = filteredData.map(d => `
                <div class="py-3 flex items-center justify-between gap-3">
                    <div class="flex items-center gap-3 min-w-0">
                        <span class="inline-block w-20 text-center py-0.5 text-xs font-bold rounded border leading-tight shrink-0 ${getCategoryBadgeClass(d.category)}">
                            ${d.category}
                        </span>
                        <div class="truncate">
                            <h5 class="text-sm font-bold text-slate-900 truncate">${d.name}</h5>
                            <p class="text-xs font-mono font-bold text-slate-600 truncate">${d.phone} ${d.vehicle ? '• ' + d.vehicle : ''}</p>
                        </div>
                    </div>
                    <button onclick="deleteContract(${d.id})" class="shrink-0 px-3 py-1.5 bg-rose-50 hover:bg-rose-600 text-rose-600 hover:text-white rounded-lg text-xs font-bold transition-all border border-rose-200 flex items-center gap-1">
                        <i class="fa-solid fa-trash-can"></i> Delete
                    </button>
                </div>
            `).join('');
        }

        // Delete Handler
        function deleteContract(id) {
            if (confirm("Are you sure you want to delete this contract?")) {
                driversData = driversData.filter(d => d.id !== id);
                renderDriversTable(driversData);
                renderAdminDeleteList();
                updateDashboardStats();
            }
        }

        // Stats Counter
        function updateDashboardStats() {
            const total = driversData.length;
            const active = driversData.filter(d => d.status === 'Active').length;

            document.getElementById('stat-total').innerText = total;
            document.getElementById('stat-active').innerText = active;
        }

        // Filter Main Table Search
        function filterDrivers() {
            const query = document.getElementById('search-input').value.toLowerCase();
            const filtered = driversData.filter(d => 
                d.name.toLowerCase().includes(query) || 
                (d.category && d.category.toLowerCase().includes(query)) ||
                d.vehicle.toLowerCase().includes(query) ||
                d.phone.includes(query)
            );
            renderDriversTable(filtered);
        }

        // Form Submit
        function handleAddDriver(e) {
            e.preventDefault();
            const name = document.getElementById('new-driver-name').value;
            const category = document.getElementById('new-driver-category').value;
            const phone = document.getElementById('new-driver-phone').value;
            const vehicle = document.getElementById('new-driver-vehicle').value.trim();

            const newEntry = {
                id: Date.now(),
                name,
                category,
                phone,
                vehicle: vehicle || "",
                status: 'Active'
            };

            driversData.push(newEntry);
            
            renderDriversTable(driversData);
            updateDashboardStats();
            renderAdminDeleteList();
            
            document.getElementById('add-driver-form').reset();
            alert('New contract entry added successfully!');
            switchTab('drivers');
        }
    </script>
